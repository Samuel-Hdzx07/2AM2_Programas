1. PUNTEROS - SINTAXIS ESENCIAL
═══════════════════════════════════════════════════════════════════════════════

DECLARACIÓN:
   int *ptr;              // Puntero a entero
   int **pptr;            // Puntero a puntero (doble)

OPERADORES:
   &variable              // Dirección de variable
   *ptr                   // Valor en la dirección (desreferencia)

INICIALIZACIÓN:
   int x = 10;
   int *ptr = &x;         // ptr apunta a x
   
ARITMÉTICA:
   ptr + 1                // Siguiente elemento
   ptr++                  // Avanzar al siguiente
   ptr < &arr[n]          // Comparar límite

CONVERSIÓN DE ACCESO:
   arr[i]    ←→    *(arr + i)
   ptr[i]    ←→    *(ptr + i)

2. ESTRUCTURAS - SINTAXIS ESENCIAL
═══════════════════════════════════════════════════════════════════════════════

DEFINICIÓN:
   struct NombreStruct {
       tipo1 campo1;
       tipo2 campo2;
   };

DECLARACIÓN:
   struct NombreStruct variable;
   struct NombreStruct *ptrStruct;

ACCESO:
   variable.campo         // Con variable normal (punto)
   ptr->campo             // Con puntero (flecha)
   (*ptr).campo           // Equivalente (menos usado)

PUNTERO A ESTRUCTURA:
   struct Dato d;
   struct Dato *ptr = &d;
   ptr->campo = valor;    // Acceso con flecha

3. PILA ESTÁTICA - ESTRUCTURA Y OPERACIONES
═══════════════════════════════════════════════════════════════════════════════

ESTRUCTURA:
   struct Pila {
       int datos[MAX];
       int tope;           // Inicia en -1
   };

INICIALIZAR:
   void inicializar(struct Pila *p) {
       p->tope = -1;
   }

ESTÁ VACÍA:
   int estaVacia(struct Pila *p) {
       return p->tope == -1;
   }

ESTÁ LLENA:
   int estaLlena(struct Pila *p) {
       return p->tope == MAX - 1;
   }

PUSH (insertar):
   int push(struct Pila *p, int valor) {
       if (p->tope == MAX - 1) return 0;  // Llena
       p->tope++;
       p->datos[p->tope] = valor;
       return 1;
   }

POP (eliminar):
   int pop(struct Pila *p, int *valor) {
       if (p->tope == -1) return 0;       // Vacía
       *valor = p->datos[p->tope];
       p->tope--;
       return 1;
   }

PEEK (consultar):
   int peek(struct Pila *p, int *valor) {
       if (p->tope == -1) return 0;
       *valor = p->datos[p->tope];
       return 1;
   }

DIAGRAMA:
   tope = 2  →  [30] ← último en entrar
                [20]
                [10] ← primero en entrar
   
   LIFO: Last In, First Out

4. COLA ESTÁTICA (CIRCULAR) - ESTRUCTURA Y OPERACIONES
═══════════════════════════════════════════════════════════════════════════════

ESTRUCTURA:
   struct Cola {
       int datos[MAX];
       int frente;         // Inicia en 0
       int final;          // Inicia en -1
       int contador;       // Inicia en 0
   };

INICIALIZAR:
   void inicializar(struct Cola *c) {
       c->frente = 0;
       c->final = -1;
       c->contador = 0;
   }

ESTÁ VACÍA:
   int estaVacia(struct Cola *c) {
       return c->contador == 0;
   }

ESTÁ LLENA:
   int estaLlena(struct Cola *c) {
       return c->contador == MAX;
   }

ENQUEUE (insertar al final):
   int enqueue(struct Cola *c, int valor) {
       if (c->contador == MAX) return 0;   // Llena
       c->final = (c->final + 1) % MAX;    // CIRCULAR
       c->datos[c->final] = valor;
       c->contador++;
       return 1;
   }

DEQUEUE (eliminar del frente):
   int dequeue(struct Cola *c, int *valor) {
       if (c->contador == 0) return 0;     // Vacía
       *valor = c->datos[c->frente];
       c->frente = (c->frente + 1) % MAX;  // CIRCULAR
       c->contador--;
       return 1;
   }

PEEK (consultar frente):
   int peek(struct Cola *c, int *valor) {
       if (c->contador == 0) return 0;
       *valor = c->datos[c->frente];
       return 1;
   }

DIAGRAMA:
   frente=0    final=2    contador=3
   
   [10] ← frente (primero en entrar)
   [20]
   [30] ← final (último en entrar)
   
   FIFO: First In, First Out

FÓRMULA CIRCULAR:
   índice = (índice + 1) % MAX
   
   Ejemplo con MAX=5:
   0, 1, 2, 3, 4, 0, 1, 2, ...  (da la vuelta)

5. TABLA COMPARATIVA: PILA vs COLA
═══════════════════════════════════════════════════════════════════════════════

┌───────────────┬─────────────────────┬─────────────────────┐
│ Concepto      │ PILA                │ COLA                │
├───────────────┼─────────────────────┼─────────────────────┤
│ Principio     │ LIFO                │ FIFO                │
│               │ Last In, First Out  │ First In, First Out │
├───────────────┼─────────────────────┼─────────────────────┤
│ Índices       │ tope                │ frente, final       │
├───────────────┼─────────────────────┼─────────────────────┤
│ Inicialización│ tope = -1           │ frente=0, final=-1  │
│               │                     │ contador=0          │
├───────────────┼─────────────────────┼─────────────────────┤
│ Insertar      │ push(tope)          │ enqueue(final)      │
│               │ tope++              │ final=(final+1)%MAX │
├───────────────┼─────────────────────┼─────────────────────┤
│ Eliminar      │ pop(tope)           │ dequeue(frente)     │
│               │ tope--              │ frente=(frente+1)%MAX│
├───────────────┼─────────────────────┼─────────────────────┤
│ Consultar     │ peek(tope)          │ peek(frente)        │
├───────────────┼─────────────────────┼─────────────────────┤
│ Vacía         │ tope == -1          │ contador == 0       │
├───────────────┼─────────────────────┼─────────────────────┤
│ Llena         │ tope == MAX-1       │ contador == MAX     │
├───────────────┼─────────────────────┼─────────────────────┤
│ Analogía      │ Pila de platos      │ Fila de banco       │
└───────────────┴─────────────────────┴─────────────────────┘

   for (int i = 0; i < c->contador; i++) {
       printf("%d", c->datos[idx]);
       idx = (idx + 1) % MAX;
   }

✦ VALIDACIÓN ANTES DE OPERACIONES:
   // Pila:
   if (!estaLlena(p)) push(p, valor);
   if (!estaVacia(p)) pop(p, &valor);

   if (!estaLlena(c)) enqueue(c, valor);
   if (!estaVacia(c)) dequeue(c, &valor);

7. ERRORES FRECUENTES - ¡EVÍTALOS!
═══════════════════════════════════════════════════════════════════════════════

❌ PUNTEROS:
   int *ptr;              // Sin inicializar - PELIGROSO
   *ptr = 5;              // ERROR: ptr no apunta a nada válido
   
   ✓ CORRECTO:
   int x;
   int *ptr = &x;
   *ptr = 5;              // OK: ptr apunta a x

❌ ESTRUCTURAS:
   struct Dato *ptr = &dato;
   ptr.campo = valor;     // ERROR: usar punto con puntero
   
   ✓ CORRECTO:
   ptr->campo = valor;    // OK: usar flecha con puntero

❌ PILA:
   push(&pila, valor);    // Sin verificar si está llena
   
   ✓ CORRECTO:
   if (!estaLlena(&pila)) {
       push(&pila, valor);
   }

❌ COLA:
   c->final = c->final + 1;      // Sin aritmética circular
   
   ✓ CORRECTO:
   c->final = (c->final + 1) % MAX;  // Con aritmética circular

❌ OLVIDOS COMUNES:
   • No incrementar/decrementar índices (tope, contador)
   • No usar % MAX en cola circular
   • Confundir frente con final
   • No verificar vacía/llena antes de operar

8. FÓRMULAS Y CÁLCULOS IMPORTANTES
═══════════════════════════════════════════════════════════════════════════════

PILA:
   • Número de elementos = tope + 1
   • Espacio disponible = MAX - tope - 1
   • Vacía: tope == -1
   • Llena: tope == MAX - 1

COLA:
   • Número de elementos = contador
   • Espacio disponible = MAX - contador
   • Vacía: contador == 0
   • Llena: contador == MAX
   • Índice circular: (índice + 1) % MAX

ARITMÉTICA DE PUNTEROS:
   • Distancia entre punteros: ptr2 - ptr1
   • Índice desde inicio: ptr - arr
   • Dirección del elemento i: arr + i o &arr[i]

9. PREGUNTAS TÍPICAS DE EXAMEN
═══════════════════════════════════════════════════════════════════════════════

Q1: ¿Cuál es la diferencia entre LIFO y FIFO?
A:  LIFO (pila): último en entrar, primero en salir
    FIFO (cola): primero en entrar, primero en salir

Q2: ¿Qué hace el operador -> en C?
A:  Accede a un miembro de estructura a través de un puntero
    ptr->campo es equivalente a (*ptr).campo

Q3: ¿Por qué usar cola circular?
A:  Para reutilizar espacios vacíos cuando el frente avanza
    Sin circular, se desperdicia memoria

Q4: ¿Cuándo está vacía una pila? ¿Y una cola?
A:  Pila vacía: tope == -1
    Cola vacía: contador == 0

Q5: ¿Cómo se calcula el siguiente índice en cola circular?
A:  índice = (índice + 1) % MAX

Q6: ¿Qué es la aritmética de punteros?
A:  Sumar/restar valores a punteros para navegar en memoria
    ptr + 1 apunta al siguiente elemento del tipo

Q7: Diferencia entre punto (.) y flecha (->)
A:  Punto (.): para variables normales de estructura
    Flecha (->): para punteros a estructura
